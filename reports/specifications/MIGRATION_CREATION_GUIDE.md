## 1. Migration이란?

**Migration**: 데이터베이스 스키마 변경 작업을 코드로 관리

### 예시: 테이블 추가

```python
# Python Model (models.py)
class Lead(Base):
    __tablename__ = "leads"
    id = Column(Integer, primary_key=True)
    name = Column(String(100))

# ↓ Alembic이 자동 감지 ↓

# Migration (versions/abc123.py)
def upgrade():
    op.create_table('leads',
        sa.Column('id', sa.Integer(), primary_key=True),
        sa.Column('name', sa.String(100))
    )

def downgrade():
    op.drop_table('leads')
```

## 5. Migration 파일 검토 및 수정

### 검토 포인트

#### 1. 테이블 생성 순서 (FK 고려)

**잘못된 순서**:
```python
def upgrade():
    # FK가 있는 테이블을 먼저 만들면 에러!
    op.create_table('inquiries',  # lead_id FK
        sa.ForeignKeyConstraint(['lead_id'], ['leads.id'])
    )
    op.create_table('leads')  # 나중에 만듦 → 에러!
```

**올바른 순서**:
```python
def upgrade():
    # 부모 테이블 먼저
    op.create_table('leads')
    # 자식 테이블 나중
    op.create_table('inquiries',
        sa.ForeignKeyConstraint(['lead_id'], ['leads.id'])
    )
```

#### 2. Default 값 확인

```python
# Alembic이 놓칠 수 있음
sa.Column('status', sa.String(20), nullable=True)

# 수동으로 default 추가
sa.Column('status', sa.String(20), server_default='new', nullable=True)
```

#### 3. Index 확인

```python
# 자주 검색하는 컬럼에 인덱스 추가
op.create_index('idx_leads_status', 'leads', ['status'])
op.create_index('idx_inquiries_lead_id', 'inquiries', ['lead_id'])
```

#### 4. NOT NULL 확인

```python
# nullable=False가 의도대로 설정되었는지 확인
sa.Column('name', sa.String(100), nullable=False)  # ✓
sa.Column('email', sa.String(255), nullable=True)  # ✓
```

### 수동 수정 예시

**자동 생성된 코드**:
```python
def upgrade() -> None:
    op.create_table('leads',
        sa.Column('id', sa.Integer(), autoincrement=True, nullable=False),
        sa.Column('name', sa.String(length=100), nullable=False),
        sa.Column('status', sa.String(length=20), nullable=True),
        sa.PrimaryKeyConstraint('id')
    )
```

**수정 후**:
```python

    op.create_table('leads',
        sa.Column('id', sa.Integer(), autoincrement=True, nullable=False),
        sa.Column('name', sa.String(length=100), nullable=False),
        sa.Column('status', sa.String(length=20), server_default='new', nullable=False),  # ← default 추가
        sa.Column('created_at', sa.DateTime(), server_default=sa.func.now(), nullable=False),  # ← now() 추가
        sa.PrimaryKeyConstraint('id')
    )

    # 인덱스 추가
    op.create_index('idx_leads_status', 'leads', ['status'])
    op.create_index('idx_leads_created_at', 'leads', ['created_at'])
```

## 11. 고급 Migration 패턴

### 패턴 1: 데이터 마이그레이션

```python
def upgrade() -> None:
    # 1. 컬럼 추가
    op.add_column('leads', sa.Column('priority', sa.String(20)))

    # 2. 기존 데이터 업데이트
    from sqlalchemy import text
    connection = op.get_bind()
    connection.execute(text("""
        UPDATE leads
        SET priority = CASE
            WHEN score >= 80 THEN 'high'
            WHEN score >= 50 THEN 'medium'
            ELSE 'low'
        END
    """))

    # 3. NOT NULL 제약 추가 (데이터 있는 상태에서)
    op.alter_column('leads', 'priority', nullable=False)
```

def upgrade() -> None:

    if 'leads' not in inspector.get_table_names():
        op.create_table('leads', ...)

### 패턴 3: 복잡한 변경 (컬럼 rename)

def upgrade() -> None:
    # Alembic은 rename을 감지 못함 → 수동 작성
